# Where this repository is linked into the home folder.
# install creates these links, replacing whatever is there, and
# `nocturne doctor` checks them. Each line is:
#
#   path in this repo    link to create
#
# A source ending in /* links every file in that folder into the target folder.
# Links that need more than this (Herdr's plugins, Zen's profile) are made by
# install itself.

# Nocturne: the palette and its generated colour files. Several apps import
# from ~/.config/nocturne.
nocturne                              ~/.config/nocturne

# Desktop: compositor, bar, launcher, notifications, volume display
hypr                                  ~/.config/hypr
waybar                                ~/.config/waybar
rofi                                  ~/.config/rofi
swaync                                ~/.config/swaync
dbus/org.freedesktop.Notifications.service ~/.local/share/dbus-1/services/org.freedesktop.Notifications.service
swayosd                               ~/.config/swayosd
quickshell                            ~/.config/quickshell
fastfetch                             ~/.config/fastfetch
# btop rewrites btop.conf, so only its theme is linked; install selects it.
btop/themes/nocturne.theme            ~/.config/btop/themes/nocturne.theme
zathura                               ~/.config/zathura

# GTK apps
gtk/gtk-3.0.css                       ~/.config/gtk-3.0/gtk.css

# Terminal and shell

starship.toml                         ~/.config/starship.toml

# bat's Nocturne theme, also used by delta and ranger's previews

# Git: shared settings only. Name, email and keys stay in ~/.gitconfig.
